import requests def check_security_headers(url): print(f"--- بدء فحص الموقع: {url} ---\n") try: # إرسال طلب HTTP للحصول على رؤوس الاستجابة response = requests.get(url, timeout=5) headers = response.headers # قائمة بأهم رؤوس الأمان التي يجب أن تتوفر في الخوادم الآمنة important_headers = { "Strict-Transport-Security": "تحمي من هجمات downgrade والتقاط البيانات (HTTPS).", "X-Content-Type-Options": "تمنع المتصفح من تخمين أنواع الملفات (MIME-sniffing).", "X-Frame-Options": "تحمي من ثغرةClickjacking (تضمين الموقع داخل iframe).", "Content-Security-Policy": "تحمي من ثغرات Cross-Site Scripting (XSS).", "X-XSS-Protection": "طبقة حماية إضافية متوافقة مع المتصفحات القديمة ضد XSS." } # التحقق من وجود كل رأس في الاستجابة found_issues = 0 for header, description in important_headers.items(): if header in headers: print(f"[✔] آمن: الرأس ({header}) موجود.") else: print(f"[✘] ثغرة/ملاحظة: الرأس ({header}) مفقود! -> {description}") found_issues += 1 print(f"\nنتيجة الفحص: تم العثور على {found_issues} ملاحظات أمنية في الإعدادات.") except requests.exceptions.RequestException as e: print(f"حدث خطأ أثناء الاتصال بالموقع: {e}") # تجربة الفحص على موقع آمن (يمكنك تغييره لأي رابط مسموح لك بفحصه) target_url = "https://example.com" check_security_headers(target_url)
